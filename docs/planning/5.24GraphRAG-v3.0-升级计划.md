### 2.4 backend/storage/graph_client.py

```python
"""Neo4j 图数据库客户端，封装 driver 生命周期和常用操作。"""
import os
from neo4j import GraphDatabase, Driver

_neo4j_driver: Driver | None = None

def _get_driver() -> Driver:
    global _neo4j_driver
    if _neo4j_driver is None:
        uri = os.getenv("NEO4J_URI", "bolt://localhost:7687")
        user = os.getenv("NEO4J_USER", "neo4j")
        password = os.getenv("NEO4J_PASSWORD", "password")
        _neo4j_driver = GraphDatabase.driver(uri, auth=(user, password))
    return _neo4j_driver

def run_cypher(query: str, params: dict = None) -> list[dict]:
    """执行只读 Cypher 查询，返回记录列表。"""
    with _get_driver().session() as session:
        result = session.run(query, params or {})
        return [dict(record) for record in result]

def write_cypher(query: str, params: dict = None) -> dict:
    """执行写入 Cypher，返回汇总信息。"""
    with _get_driver().session() as session:
        result = session.run(query, params or {})
        summary = result.consume()
        return {
            "nodes_created": summary.counters.nodes_created,
            "relationships_created": summary.counters.relationships_created,
            "properties_set": summary.counters.properties_set,
        }

def close_driver():
    global _neo4j_driver
    if _neo4j_driver:
        _neo4j_driver.close()
        _neo4j_driver = None
```

### 3.3 backend/documents/graph_extractor.py

```python
"""图谱三元组抽取器。利用 LLM 结构化输出从 L2 文本提取实体关系。"""
import json, os, asyncio
from typing import List
from pydantic import BaseModel, Field
from langchain_core.messages import SystemMessage, HumanMessage
from backend.schemas import GraphEntity, GraphRelation


class ExtractionResult(BaseModel):
    entities: List[GraphEntity] = Field(default_factory=list)
    relations: List[GraphRelation] = Field(default_factory=list)

EXTRACTION_PROMPT = """你是一个知识图谱构建专家。请从以下文本中提取实体和关系。

规则:
1. 实体: 提取文本中出现的所有重要名词——包含人名、机构、技术、概念、产品、模型、算法、数据、方法。
2. 关系: 识别实体之间的语义关系。
3. 消歧: 如果多个名称指向同一实体，使用最规范的名称。
4. 只提取文本中明确出现的信息，不要编造。

输出 JSON 格式:
{
  "entities": [
    {"name": "实体名", "type": "Person|Organization|Technology|Concept|Model|Method|Data", "description": "简短描述"}
  ],
  "relations": [
    {"subject": "源实体", "predicate": "DEPENDS_ON|CONTAINS|CITES|USES|PART_OF|PROPOSES|EVALUATES", "object": "目标实体", "description": "关系描述", "weight": 0.8}
  ]
}
"""

MAX_ENTITIES_PER_CHUNK = 20
MAX_RELATIONS_PER_CHUNK = 30
MAX_CONCURRENT_EXTRACTIONS = 5


async def extract_from_l2_chunk(text: str, filename: str) -> ExtractionResult:
    """从单个 L2 文本块提取实体关系。"""
    import time
    from backend.agent.orchestrator import _get_worker_model

    model = _get_worker_model()
    prompt = f"文件名: {filename}\n\n文本:\n{text[:1500]}"  # 截断，避免 Token 溢出
    messages = [
        SystemMessage(content=EXTRACTION_PROMPT),
        HumanMessage(content=prompt),
    ]

    try:
        structured_model = model.with_structured_output(ExtractionResult)
        result = structured_model.invoke(messages)
        # 限制数量
        result.entities = result.entities[:MAX_ENTITIES_PER_CHUNK]
        result.relations = result.relations[:MAX_RELATIONS_PER_CHUNK]
        return result
    except Exception as e:
        print(f"[EXTRACT] LLM 调用失败: {e}")
        return ExtractionResult()


async def extract_from_l2_chunks(
    l2_chunks: List[dict], filename: str, progress_callback=None
) -> ExtractionResult:
    """从一批 L2 块并发提取，合并去重。"""
    semaphore = asyncio.Semaphore(MAX_CONCURRENT_EXTRACTIONS)

    async def _extract_one(chunk):
        async with semaphore:
            return await extract_from_l2_chunk(chunk["text"], filename)

    tasks = [_extract_one(c) for c in l2_chunks]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    # 合并 + 去重
    all_entities = {}
    all_relations = []
    for r in results:
        if isinstance(r, Exception):
            continue
        for e in r.entities:
            all_entities[e.name] = e  # 同名实体后者覆盖
        all_relations.extend(r.relations)

    return ExtractionResult(
        entities=list(all_entities.values()),
        relations=all_relations,
    )
```

### 3.4 backend/storage/graph_ingestion.py

```python
"""将抽取结果写入 Neo4j。"""
from .graph_client import write_cypher

def ingest_extraction_result(
    entities: list, relations: list, l3_chunk_ids: list[str]
) -> dict:
    """批量写入实体和关系到 Neo4j。"""
    stats = {"entities": 0, "relations": 0}

    # 写入实体
    for entity in entities:
        try:
            write_cypher(
                """
                MERGE (e:Entity {name: $name})
                ON CREATE SET e.type = $type, e.description = $desc
                ON MATCH SET e.type = $type,
                    e.description = CASE WHEN $desc <> ''
                        THEN $desc ELSE e.description END
                """,
                {"name": entity.name, "type": entity.type, "desc": entity.description},
            )
            stats["entities"] += 1
        except Exception as e:
            print(f"[INGEST] Entity write error: {e}")

    # 写入关系
    for rel in relations:
        try:
            write_cypher(
                """
                MATCH (a:Entity {name: $subject})
                MATCH (b:Entity {name: $object})
                MERGE (a)-[r:RELATES_TO {type: $predicate}]->(b)
                ON CREATE SET
                    r.description = $desc,
                    r.weight = $weight,
                    r.source_chunks = $chunks
                ON MATCH SET
                    r.weight = CASE WHEN $weight > r.weight THEN $weight ELSE r.weight END,
                    r.source_chunks = apoc.coll.toSet(
                        coalesce(r.source_chunks, []) + $chunks
                    )
                """,
                {
                    "subject": rel.subject,
                    "object": rel.object,
                    "predicate": rel.predicate,
                    "desc": rel.description,
                    "weight": rel.weight,
                    "chunks": rel.source_chunks or l3_chunk_ids,
                },
            )
            stats["relations"] += 1
        except Exception as e:
            print(f"[INGEST] Relation write error: {e}")

    print(f"[INGEST] Written: {stats}")
    return stats
```

### 4.3 backend/graph/community.py

```python
"""社区聚类与摘要生成。"""
import networkx as nx
import community as community_louvain  # python-louvain
from backend.storage.graph_client import run_cypher, write_cypher
from backend.milvus.client import MilvusManager
from backend.embedding.service import EmbeddingService


def build_graph_from_neo4j() -> nx.Graph:
    """从 Neo4j 拉取全量图。"""
    rows = run_cypher(
        "MATCH (a:Entity)-[r:RELATES_TO]->(b:Entity) "
        "RETURN a.name AS source, b.name AS target, r.weight AS weight"
    )
    G = nx.Graph()
    for row in rows:
        weight = row.get("weight", 0.5) if row.get("weight") else 0.5
        G.add_edge(row["source"], row["target"], weight=float(weight))
    # 孤立节点（没有关系的实体）
    isolated = run_cypher(
        "MATCH (e:Entity) WHERE NOT (e)-[:RELATES_TO]-() RETURN e.name AS name"
    )
    for row in isolated:
        G.add_node(row["name"])
    return G


def run_leiden_clustering(G: nx.Graph, resolution: float = 1.0) -> dict[str, int]:
    """运行 Louvain 聚类（Leiden 近似），返回 {node_name: community_id}。"""
    partition = community_louvain.best_partition(G, resolution=resolution, random_state=42)
    return partition


def write_community_ids(partition: dict[str, int]):
    """将社区 ID 写回 Neo4j 实体节点。"""
    batch = []
    for name, cid in partition.items():
        batch.append({"name": name, "cid": cid})

    # 分批写入
    for i in range(0, len(batch), 100):
        chunk = batch[i:i + 100]
        write_cypher(
            "UNWIND $batch AS item "
            "MATCH (e:Entity {name: item.name}) "
            "SET e.community_id = toString(item.cid)",
            {"batch": chunk},
        )
    print(f"[CLUSTER] Written community IDs for {len(batch)} entities")


def generate_community_summary(community_id: str) -> str:
    """为一个社区生成综述摘要。"""
    from backend.agent.orchestrator import _get_worker_model

    # 收集社区上下文
    rows = run_cypher(
        "MATCH (e:Entity {community_id: $cid}) "
        "OPTIONAL MATCH (e)-[r:RELATES_TO]-(other:Entity) "
        "RETURN e.name AS entity, e.type AS type, e.description AS desc, "
        "collect(DISTINCT {rel: type(r), target: other.name}) AS relations",
        {"cid": community_id},
    )

    # 拼上下文
    context_parts = []
    for row in rows:
        context_parts.append(f"- [{row['type']}] {row['entity']}: {row.get('desc', '')}")
    context = "\n".join(context_parts)

    PROMPT = f"""你是知识分析师。以下是知识图谱中一个社区的实体列表。请写一份简短的综述报告（200-400 字），总结该社区涵盖的主要主题、关键实体及其关联。

社区实体:
{context}

综述报告:"""

    model = _get_worker_model()
    response = model.invoke([HumanMessage(content=PROMPT)])
    return response.content


def generate_all_summaries(partition: dict[str, int]) -> list[dict]:
    """为所有社区生成摘要，返回 [{community_id, summary_text}]。"""
    unique_cids = set(partition.values())
    summaries = []
    for i, cid in enumerate(unique_cids):
        print(f"[SUMMARY] Generating for community {cid} ({i+1}/{len(unique_cids)})")
        text = generate_community_summary(str(cid))
        summaries.append({"community_id": str(cid), "summary_text": text})
    return summaries


def index_summaries_to_milvus(summaries: list[dict]):
    """将社区摘要向量化并写入 Milvus。"""
    embed_service = EmbeddingService()
    milvus = MilvusManager()

    # 确保 Collection 存在（新建或复用）
    # 简化: 使用现有 collection 或创建 community_summaries collection
    texts = [s["summary_text"] for s in summaries]
    embeddings = embed_service.get_embeddings(texts)

    insert_data = [
        {
            "dense_embedding": emb,
            "sparse_embedding": {},  # BM25 稀疏向量（可后续补）
            "text": s["summary_text"],
            "filename": f"community_{s['community_id']}",
            "file_type": "CommunitySummary",
            "page_number": 0,
            "chunk_idx": 0,
            "chunk_id": f"community_{s['community_id']}",
            "parent_chunk_id": "",
            "root_chunk_id": "",
            "chunk_level": 0,
        }
        for s, emb in zip(summaries, embeddings)
    ]
    milvus.init_collection()
    milvus.insert(insert_data)
    print(f"[SUMMARY] Indexed {len(summaries)} summaries to Milvus")
```

### 5.2 局部搜索（Local Search）

```python
# backend/rag/graph_retriever.py

def local_graph_search(query: str, top_k: int = 5, graph_hops: int = 2) -> dict:
    """
    局部图搜索:
    1. Milvus 混合检索 → Top-K 文本块
    2. 提取块中关联的 Neo4j 实体 → 向外扩散
    3. 合并 CHUNKS + GRAPH TRIPLES → 返回
    """
    from backend.rag.utils import retrieve_documents
    from backend.storage.graph_client import run_cypher

    # Step 1: Vector search
    retrieved = retrieve_documents(query, top_k=top_k)
    chunks = retrieved.get("docs", [])

    # Step 2: Graph expansion
    chunk_ids = [c.get("chunk_id", "") for c in chunks if c.get("chunk_id")]
    graph_triples = []

    if chunk_ids:
        # 找到关联这些 chunk 的关系边
        cypher = """
        MATCH (a:Entity)-[r:RELATES_TO]->(b:Entity)
        WHERE any(cid IN r.source_chunks WHERE cid IN $chunk_ids)
        RETURN a.name AS subject, type(r) AS predicate, b.name AS object,
               r.description AS desc, r.weight AS weight
        LIMIT 30
        """
        triples = run_cypher(cypher, {"chunk_ids": chunk_ids})
        graph_triples.extend(triples)

        # 外扩 1-hop: 捞取关联实体的邻居
        if graph_hops >= 1:
            entity_names = set()
            for t in triples:
                entity_names.add(t["subject"])
                entity_names.add(t["object"])

            if entity_names:
                neighbor_cypher = """
                MATCH (e:Entity)-[r:RELATES_TO]-(other:Entity)
                WHERE e.name IN $names AND other.name NOT IN $names
                RETURN e.name AS source, type(r) AS predicate,
                       other.name AS target, r.description AS desc
                LIMIT 20
                """
                neighbors = run_cypher(neighbor_cypher, {"names": list(entity_names)})
                graph_triples.extend(neighbors)

    # Step 3: Merge context
    chunk_texts = [f"[Chunk {i+1}] {c.get('text', '')[:300]}" for i, c in enumerate(chunks)]
    triple_texts = [
        f"({t['subject']})-[{t.get('predicate','')}]->({t.get('target','')}): {t.get('desc','')}"
        for t in graph_triples
    ]

    context = "\n\n".join(
        ["## 检索到的文本片段", *chunk_texts, "## 知识图谱关系", *triple_texts]
    )

    return {
        "chunks": chunks,
        "graph_triples": graph_triples,
        "context": context,
        "mode": "local_graph",
    }
```

### 5.4 三通道 RRF 融合

```python
# 在 backend/rag/utils.py 中修改 _rrf_fusion 或新增

def rrf_fusion_three_channel(
    dense_results: list,    # [(doc, score), ...]
    sparse_results: list,   # [(doc, score), ...]
    graph_results: list,    # [(doc, graph_score), ...]
    k: int = 60,
    weights: tuple = (0.4, 0.3, 0.3),
    top_k: int = 10,
) -> list:
    """
    三通道 RRF 融合: Dense + Sparse + Graph.
    graph_score 基于实体在图中的 PageRank 中心度。
    """
    w1, w2, w3 = weights
    scores = {}

    for rank, (doc, _) in enumerate(dense_results, 1):
        key = doc.get("chunk_id") or doc.get("text", "")[:50]
        scores[key] = scores.get(key, 0) + w1 / (k + rank)

    for rank, (doc, _) in enumerate(sparse_results, 1):
        key = doc.get("chunk_id") or doc.get("text", "")[:50]
        scores[key] = scores.get(key, 0) + w2 / (k + rank)

    for rank, (doc, _) in enumerate(graph_results, 1):
        key = doc.get("chunk_id") or doc.get("text", "")[:50]
        scores[key] = scores.get(key, 0) + w3 / (k + rank)

    # 按最终得分排序
    all_docs = {**{d.get("chunk_id") or d.get("text","")[:50]: d for d, _ in dense_results},
                **{d.get("chunk_id") or d.get("text","")[:50]: d for d, _ in sparse_results},
                **{d.get("chunk_id") or d.get("text","")[:50]: d for d, _ in graph_results}}

    ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return [all_docs[k] for k, _ in ranked[:top_k]]
```
